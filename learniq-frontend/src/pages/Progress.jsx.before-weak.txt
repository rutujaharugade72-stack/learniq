import { useEffect, useState } from "react";
import { Link } from "react-router-dom";
import {
  ArrowLeft,
  TrendingUp,
  Award,
  Target,
  Clock,
  BookOpen,
  CheckCircle2,
  AlertTriangle,
  RotateCcw,
  Sparkles,
  Flame,
  Activity,
  Calendar,
} from "lucide-react";
import { useAuth } from "../context/AuthContext";
import { useReportActivity } from "../context/AssistantContext";

function accuracyTone(rate) {
  if (rate >= 0.75) return { bg: "#dcfce7", border: "#22c55e", text: "#15803d", label: "Mastered" };
  if (rate >= 0.5) return { bg: "#fef3c7", border: "#f59e0b", text: "#b45309", label: "Improving" };
  return { bg: "#fee2e2", border: "#ef4444", text: "#b91c1c", label: "Needs Practice" };
}

export default function Progress() {
  const { token, user } = useAuth();
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");
  const [stats, setStats] = useState({
    totalAttempts: 0,
    overallAccuracy: 0,
    topics: [],
    recentActivity: [],
  });

  useReportActivity({
    page: "Progress & Mastery",
    summary: `Student is tracking subject mastery: ${stats.topics.length} subjects recorded, ${stats.overallAccuracy}% overall accuracy.`,
  });

  useEffect(() => {
    async function loadData() {
      try {
        // Fetch quiz performance/weak spot telemetry
        const resAnalytics = await fetch("/api/why-stuck/me", {
          headers: { Authorization: `Bearer ${token}` },
        });
        const dataAnalytics = resAnalytics.ok ? await resAnalytics.json() : null;

        // Fetch recent learning activity
        const resActivity = await fetch("/api/activity", {
          headers: { Authorization: `Bearer ${token}` },
        });
        const dataActivity = resActivity.ok ? await resActivity.json() : [];

        if (dataAnalytics) {
          const totalAtt = dataAnalytics.total_attempts || 0;
          let correctAtt = 0;
          const topicList = (dataAnalytics.topics || []).map((t) => {
            const subtopics = t.subtopic_breakdown || [];
            const avgAcc = subtopics.length
              ? subtopics.reduce((acc, cur) => acc + (cur.accuracy || 0), 0) / subtopics.length
              : t.accuracy || 0;
            return {
              name: t.topic,
              accuracy: Math.round(avgAcc * 100),
              subtopics: subtopics.map((s) => ({
                name: s.subtopic,
                accuracy: Math.round((s.accuracy || 0) * 100),
                attempts: s.attempts || 0,
              })),
            };
          });

          const overall = topicList.length
            ? Math.round(topicList.reduce((acc, cur) => acc + cur.accuracy, 0) / topicList.length)
            : 0;

          setStats({
            totalAttempts: totalAtt,
            overallAccuracy: overall,
            topics: topicList,
            recentActivity: Array.isArray(dataActivity) ? dataActivity.slice(0, 8) : [],
          });
        }
      } catch (err) {
        setError(err.message || "Failed to load progress telemetry");
      } finally {
        setLoading(false);
      }
    }
    loadData();
  }, [token]);

  return (
    <div className="td-page" style={{ maxWidth: 1080, margin: "0 auto" }}>
      {/* Back link */}
      <Link to="/" className="td-back">
        <ArrowLeft size={16} /> Back to dashboard
      </Link>

      {/* Yellow Sticker Hero Banner */}
      <section className="td-hero">
        <div className="td-hero-text">
          <span className="td-badge-sticker">
            <TrendingUp size={14} /> Learning Intelligence
          </span>
          <h2>My Progress & Mastery</h2>
          <p>
            Real metrics computed from actual quizzes and tests. Track your learning curve
            without shortcuts.
          </p>
        </div>
        <div className="td-badge-sticker" style={{ background: "#facc15" }}>
          {stats.overallAccuracy}% Overall Mastery
        </div>
      </section>

      {/* Quick KPI Bento Cards */}
      <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fit, minmax(220px, 1fr))", gap: 16 }}>
        <div className="td-card" style={{ display: "flex", alignItems: "center", gap: 14 }}>
          <div style={{ padding: 12, background: "#fef08a", border: "2px solid #000", borderRadius: 12 }}>
            <Award size={26} color="#000" />
          </div>
          <div>
            <div style={{ fontSize: "0.8rem", fontWeight: 800, color: "var(--muted, #64748b)" }}>
              TOTAL QUESTIONS TRIED
            </div>
            <div style={{ fontSize: "1.75rem", fontWeight: 900, color: "var(--text, #000)" }}>
              {stats.totalAttempts}
            </div>
          </div>
        </div>

        <div className="td-card" style={{ display: "flex", alignItems: "center", gap: 14 }}>
          <div style={{ padding: 12, background: "#bbf7d0", border: "2px solid #000", borderRadius: 12 }}>
            <Target size={26} color="#000" />
          </div>
          <div>
            <div style={{ fontSize: "0.8rem", fontWeight: 800, color: "var(--muted, #64748b)" }}>
              SUBJECTS TRACKED
            </div>
            <div style={{ fontSize: "1.75rem", fontWeight: 900, color: "var(--text, #000)" }}>
              {stats.topics.length}
            </div>
          </div>
        </div>

        <div className="td-card" style={{ display: "flex", alignItems: "center", gap: 14 }}>
          <div style={{ padding: 12, background: "#bae6fd", border: "2px solid #000", borderRadius: 12 }}>
            <Flame size={26} color="#000" />
          </div>
          <div>
            <div style={{ fontSize: "0.8rem", fontWeight: 800, color: "var(--muted, #64748b)" }}>
              STUDY CONSISTENCY
            </div>
            <div style={{ fontSize: "1.75rem", fontWeight: 900, color: "var(--text, #000)" }}>
              {stats.recentActivity.length > 3 ? "Active 🔥" : "Building ⚡"}
            </div>
          </div>
        </div>
      </div>

      {/* Main Grid: Subject Breakdown vs Recent Activity */}
      <div style={{ display: "grid", gridTemplateColumns: "1.3fr 0.9fr", gap: 20 }}>
        {/* Left Column: Subject Mastery Cards */}
        <div className="td-card" style={{ display: "flex", flexDirection: "column", gap: 18 }}>
          <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center" }}>
            <h3 className="td-card-title" style={{ margin: 0 }}>
              <BookOpen size={18} /> Subject Accuracy & Mastery
            </h3>
            <Link to="/quiz" className="td-btn small" style={{ textDecoration: "none" }}>
              Take New Test
            </Link>
          </div>

          {loading ? (
            <div style={{ textAlign: "center", padding: "2rem", color: "var(--muted, #64748b)" }}>
              Analyzing quiz telemetry...
            </div>
          ) : stats.topics.length === 0 ? (
            <div style={{ padding: "2rem 1rem", textAlign: "center", color: "var(--muted, #64748b)" }}>
              <p style={{ margin: "0 0 1rem", fontWeight: 700 }}>No subject tests completed yet.</p>
              <Link to="/quiz" className="td-btn small" style={{ textDecoration: "none" }}>
                Generate your first quiz →
              </Link>
            </div>
          ) : (
            <div style={{ display: "flex", flexDirection: "column", gap: 16 }}>
              {stats.topics.map((t) => {
                const tone = accuracyTone(t.accuracy / 100);
                return (
                  <div
                    key={t.name}
                    style={{
                      border: "2px solid var(--border, #000)",
                      borderRadius: 14,
                      padding: "14px 18px",
                      background: "var(--surface, #ffffff)",
                    }}
                  >
                    <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: 8 }}>
                      <strong style={{ fontSize: "1.05rem", color: "var(--text, #000)" }}>{t.name}</strong>
                      <span
                        style={{
                          fontSize: "0.78rem",
                          fontWeight: 800,
                          padding: "3px 8px",
                          borderRadius: 999,
                          border: `1.5px solid ${tone.border}`,
                          background: tone.bg,
                          color: tone.text,
                        }}
                      >
                        {t.accuracy}% · {tone.label}
                      </span>
                    </div>

                    <div
                      style={{
                        height: 10,
                        width: "100%",
                        background: "var(--bg, #f1f5f9)",
                        border: "1.5px solid var(--border, #cbd5e1)",
                        borderRadius: 6,
                        overflow: "hidden",
                        marginBottom: 10,
                      }}
                    >
                      <div
                        style={{
                          width: `${t.accuracy}%`,
                          height: "100%",
                          background: t.accuracy >= 75 ? "#10b981" : t.accuracy >= 50 ? "#f59e0b" : "#ef4444",
                          borderRadius: 4,
                          transition: "width 0.4s ease",
                        }}
                      />
                    </div>

                    {/* Subtopic Chips */}
                    {t.subtopics.length > 0 && (
                      <div style={{ display: "flex", flexWrap: "wrap", gap: 6 }}>
                        {t.subtopics.map((s) => (
                          <span
                            key={s.name}
                            style={{
                              fontSize: "0.75rem",
                              fontWeight: 700,
                              padding: "2px 8px",
                              borderRadius: 6,
                              background: "var(--bg, #f8fafc)",
                              border: "1px solid var(--border, #cbd5e1)",
                              color: "var(--muted, #475569)",
                            }}
                          >
                            {s.name}: {s.accuracy}%
                          </span>
                        ))}
                      </div>
                    )}
                  </div>
                );
              })}
            </div>
          )}
        </div>

        {/* Right Column: Recent Activity Feed */}
        <div className="td-card" style={{ display: "flex", flexDirection: "column", gap: 16 }}>
          <h3 className="td-card-title" style={{ margin: 0 }}>
            <Activity size={18} /> Verified Learning Logs
          </h3>

          {stats.recentActivity.length === 0 ? (
            <p style={{ fontSize: "0.88rem", color: "var(--muted, #64748b)" }}>
              No completed sessions recorded yet today.
            </p>
          ) : (
            <div style={{ display: "flex", flexDirection: "column", gap: 10 }}>
              {stats.recentActivity.map((act, i) => (
                <div
                  key={act._id || i}
                  style={{
                    display: "flex",
                    alignItems: "flex-start",
                    gap: 10,
                    padding: "10px 12px",
                    background: "var(--bg, #f8fafc)",
                    border: "1.5px solid var(--border, #e2e8f0)",
                    borderRadius: 10,
                  }}
                >
                  <CheckCircle2 size={16} color="#10b981" style={{ flexShrink: 0, marginTop: 2 }} />
                  <div style={{ minWidth: 0, flex: 1 }}>
                    <div style={{ fontSize: "0.85rem", fontWeight: 800, color: "var(--text, #1e293b)" }}>
                      {act.title}
                    </div>
                    {act.description && (
                      <div style={{ fontSize: "0.75rem", color: "var(--muted, #64748b)" }}>
                        {act.description}
                      </div>
                    )}
                  </div>
                </div>
              ))}
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
